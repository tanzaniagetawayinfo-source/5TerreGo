Configurazione Supabase

Crea una tabella trails per salvare i sentieri. Il campo route_geojson contiene la linea precisa del percorso, mentre points contiene i punti inseriti tenendo premuto sulla mappa.

create table if not exists public.trails ( id uuid primary key default gen_random_uuid(), created_at timestamptz default now(), updated_at timestamptz default now(), name text not null, description text, difficulty text, estimated_duration text, start_name text, end_name text, distance_km numeric, points jsonb not null default '[]'::jsonb, route_geojson jsonb, owner_email text ); alter table public.trails enable row level security; grant select on public.trails to anon; grant select, insert, update, delete on public.trails to authenticated;
